from dataclasses import dataclass, field


@dataclass
class Slot:

    expert: int = -1
    ready: bool = False
    pinned: int = 0
    score: float = 0.0
    last_use: int = 0


@dataclass
class SlotCache:
    n_slots: int
    half_life: float = 36.0
    slots: list = field(default_factory=list)
    index: dict = field(default_factory=dict)
    clock: int = 0
    hits: int = 0

        self.clock += 1

    def _touch(self, slot):
        age = self.clock - slot.last_use
        slot.score = slot.score * math.exp(-self.decay * age) + 1.0
        slot.last_use = self.clock

        best = None
        best_score = float("inf")
        for i, slot in enumerate(self.slots):
            if slot.pinned or not slot.ready:
                continue
            age = self.clock - slot.last_use
            score = slot.score * math.exp(-self.decay * age)
            if score < best_score:
                best_score = score
                best = i
        return best

        i = self.index.get(key)
        if i is not None:
            self.hits += 1
            self._touch(self.slots[i])
            return i, True
        self.misses += 1
        if self.free:
            i = self.free.pop()
        else:

        slot.ready = False
        slot.score = 0.0

        self.index[key] = i
        return i, False

        self.slots[i].ready = True

        self.slots[i].pinned += 1

        total = self.hits + self.misses
        return self.hits / total if total else 0.0
